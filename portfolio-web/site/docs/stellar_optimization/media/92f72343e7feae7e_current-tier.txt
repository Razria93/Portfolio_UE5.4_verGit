48: EAIRuntimeLODTier FAIRuntimeLODTierResolver::ResolveTier(const FAIRuntimeLODTierContext& InContext)
49: {
50: 	if (InContext.bDormantCandidate) return EAIRuntimeLODTier::Dormant;
51: 	if (IsAlwaysCombatCriticalIntentState(InContext.AIIntentState)) return EAIRuntimeLODTier::CombatCritical;
52: 
53: 	switch (InContext.CombatRole)
54: 	{
55: 	case ECombatRole::Engage:
56: 		return EAIRuntimeLODTier::CombatCritical;
57: 
58: 	case ECombatRole::Alert:
59: 		return EAIRuntimeLODTier::CombatSupport;
60: 
61: 	case ECombatRole::Observe:
62: 		return EAIRuntimeLODTier::Awareness;
63: 
64: 	case ECombatRole::None:
65: 	default:
66: 		return InContext.bHasTargetAwareness ? EAIRuntimeLODTier::Awareness : EAIRuntimeLODTier::Background;
67: 	}
68: }
